## §0 범위·전제

- 범위: 4단계 로드맵의 **1단계(DB+파이프라인) + 3단계(대시보드)**. 2단계(마이크로 챌린지 UI)는 별도 발주.
- 대시보드 = **Expo 확정**(유호 08-06 "expo로 해줘"): SYNK-talk 앱 안의 검수자 역할 화면, PC에서는 Expo web 빌드로 사용. 한 코드베이스.
- 말하기 과제 = **20~30초**(유호 확정). 검수 = 저신뢰 구간 + 무작위 5% 감사 레인.
- ⛔ **수집 개시 게이트**: 동의 6항목(음성·얼굴영상·교실녹음·국외이전·제3자 처리·철회 시 기학습 유지) 개정 **후**에만 실데이터 수집 시작 — 개정 전 수집분은 엔진에 못 쓴다(소급 불가). 문구 초안 = 클로드 몫(별도).
- Gemini = **유료 티어 필수**(무료 티어는 구글 학습에 사용 — 학생 데이터 금지). 모델 = 현행 flash(3.6 계열).

## §1 DDL v2 (그대로 실행 가능)

```sql
-- 1) 학생 제출 과제 (유호 원안 + 배선·텔레메트리·수리 반영)
create table student_homeworks (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null,               -- FK: L0 learners(id) [CHK-1: L0_스키마.sql의 실제 테이블·컬럼명 대조 후 제약 추가]
  homework_type varchar(50) not null,     -- 'audio' | 'text' | 'essay_image'
  task_type varchar(20) not null,         -- 'read_aloud' | 'response' | 'free' | 'interview' | 'style_shift' | 'exam_review'  (배선③ — 낭독/자유 혼합 금지의 근거 태그)
  challenge_id varchar(50),               -- 마이크로 챌린지 식별(교수님멘탈·보고서교정·알바변명…)
  -- ↓ §8 채택분 4필드 — 전부 「그때만 알 수 있는 값」이라 나중에 못 붙인다
  level_snapshot text,                    -- 제출 시점 급수. 🔴소급 절대 불가 — 1년 뒤 급수가 오르면 과거에 몇 급이었는지 재구성 불능(난이도 정규화·성장 곡선의 유일한 기준선)
  addressee_level varchar(10),            -- 'banmal'|'haeyo'|'hapsyo' — style_shift 과제 전용. 없으면 같은 내용 녹음 3건이 서로 구분 안 돼 높임법 데이터가 통째로 죽는다
  target_phonemes text[],                 -- 이 과제가 겨냥한 음소('ㅃ','ㅡ','ㅢ','받침ㄹ'…). 과제 설계 시점에만 아는 값 — 음소별 오류율 집계의 키
  retry_of_homework_id uuid references student_homeworks(id) on delete set null,  -- 교정 받고 다시 낸 고리. 「학습이 실제로 일어났는가」의 유일한 신호(원문→교정→재작성 3단 원칙 승계)
  prompt_seed text,                       -- 매회 무작위 문장(규칙④ 제안 반영 — 대리 제출·재사용 오염 방어. 유호 미확정 시 미사용 가능, 칼럼은 유지)
  file_url text,
  duration_sec numeric(6,2),
  privacy_class varchar(20) not null default 'voice_high',   -- c4 계약 privacy_class 축 준수
  self_reported_tension smallint,         -- 1~5, interview 과제만 (배선② — 나중에 못 붙인다)
  -- 무의식 텔레메트리 (유호 4·5신 — JSONB 채택: 챌린지마다 지표가 달라도 스키마 변경 없이 수용)
  telemetry jsonb,                        -- 예: {"first_key_delay_ms":420,"backspace_typos":3,"chunk_insert_count":0,
                                          --      "hints":[{"t_ms":1200,"type":"word|structure|mn_explain","stuck_ms":3400,"recovered":true}],
                                          --      "item_dwell_ms":{"q7":41200}}
                                          -- 🔑 힌트는 「눌렀다」가 아니라 **「막힌 시간 → 어떤 힌트 → 다시 말했는가」 3종 세트**로 남긴다(§8 A-4·C-3).
                                          --    type 없이 클릭 수만 세면 「어떤 힌트가 듣는가」를 영영 못 배운다 — 스캐폴딩 학습의 라벨이 곧 이 세 값이다.
                                          --    item_dwell_ms = 문항별 체류(TOPIK 타임아웃 구간 · 8신 H-6)
                                          -- 🔴 측정 안 된 키는 0이 아니라 「생략」 — 0 디폴트는 「측정 없음」과 「실측 0」을 영영 못 가른다(5신 원안의 0 디폴트 기각)
                                          -- chunk_insert_count>0 = 타자 지표 신뢰 하락 표식(추천단어 탭은 자모가 안 남는다)
  -- 파생 캐시(원천=stt_segments, 언제든 재계산 가능)
  speech_wpm numeric(5,2),
  pause_duration_sec numeric(5,2),
  status varchar(20) not null default 'pending',  -- pending | processing | ai_processed | verified | discarded
  retry_count integer not null default 0,
  is_audit_sample boolean not null default (random() < 0.05),  -- 무작위 5% 감사 레인(고신뢰 오전사는 이 표본으로만 잡힌다)
  created_at timestamptz not null default now()
);

-- 2) AI 1차 처리 로그 (한수더: 1차안+버전 영구 보존 — verified와 diff가 파인튜닝 쌍)
create table ai_processing_logs (
  id uuid primary key default gen_random_uuid(),
  homework_id uuid not null references student_homeworks(id) on delete cascade,
  stt_model_version text not null,        -- 예: 'whisper-1'
  llm_model_version text not null,        -- 예: 'gemini-flash-3.6'
  prompt_version text not null,           -- 예: 'correct-v1'
  stt_raw_text text,
  stt_segments jsonb,                     -- verbose_json: [{start,end,text,avg_logprob,no_speech_prob}] — 무음 하이라이트·구간 지정·검수 우선순위의 원천
  stt_avg_logprob numeric(6,3),           -- 세그먼트 평균(낮을수록 저신뢰 → 검수 우선)
  ai_corrected_text text,
  detected_error_types text[],            -- 복수 라벨(한 문장이 직역+문법 동시 가능). enum 값은 c5 계약에서 확정 [CHK-2: 레코드 5형에서 파생 — 새 축 발명 금지]
  l1_source_phrase_draft text,            -- 직역 오류 시 원인 몽골어 관용구 Gemini 1차 추정(5신 초격차 스펙 — 검수자는 확인만, 타이핑 0)
  mongolian_explanation_draft text,
  processed_at timestamptz not null default now()
);

-- 3) 검수 완료 학습셋 (회사 금고 — 연쇄삭제 금지)
create table verified_learning_datasets (
  id uuid primary key default gen_random_uuid(),
  homework_id uuid not null references student_homeworks(id) on delete restrict,
  ai_log_id uuid not null references ai_processing_logs(id) on delete restrict,
  verified_student_text text not null,    -- 「들린 대로」 — 학생 오류 보존(파인튜닝 입력). Whisper의 '고쳐 듣기'를 사람이 되돌린 값
  verified_corrected_text text not null,  -- 교정문(파인튜닝 출력)
  final_mongolian_explanation text,
  error_types text[] not null default '{}',
  l1_source_phrase text,                  -- 확인된 원인 몽골어 관용구(직역 오류만) — L1 간섭 1:1 매핑의 키(5신)
  code_switch_spans jsonb,                -- [{start_ms,end_ms,mn_text}] (За·Тийм 보정) = 5신 code_switching 스펙과 동일
  rubric_scores jsonb,                    -- essay_image·쓰기 과제만: {"내용":12,"전개":8,"어휘문법":9,"감점":["격식체 혼용"],"예상등급":"4급"}
                                          -- 🔴검수자(강사)만 매길 수 있는 값 — 나중에 붙이려면 강사를 다시 앉혀야 한다(TOPIK 쓰기 첨삭 엔진의 유일한 정답지)
  was_edited boolean not null,            -- false=Enter 무편집 승인 / true=수정 승인(고가치 표본)
  listened_ms integer not null default 0, -- 검수자 실제 청취 시간 — 검수자 1명 체제의 품질 계기판
  inspector_id uuid not null,
  verified_at timestamptz not null default now()
);

-- 4) 접속 0도 기록 (배선④ — 부재는 이벤트가 아니다)
create table daily_activity (
  student_id uuid not null,
  activity_date date not null,
  session_count integer not null default 0,
  primary key (student_id, activity_date)
);
-- pg_cron(Supabase 확장)으로 매일 1회: 재적 전원에 대해 그날 행 upsert(활동 없으면 0) [CHK-1의 재적 명단 소스 확정 후 잡 등록]

create index on student_homeworks (status, created_at);
create index on student_homeworks (student_id, created_at);
create index on ai_processing_logs (homework_id);
create index on verified_learning_datasets (homework_id);

alter table student_homeworks enable row level security;
alter table ai_processing_logs enable row level security;
alter table verified_learning_datasets enable row level security;
alter table daily_activity enable row level security;
-- 정책 의도(정책문은 [CHK-3] L0 §4 인증 구조와 함께 작성):
--   학생: 자기 행 insert/select만 · 검수자 role: 전체 select + verified insert · n8n: service_role 키
-- 스토리지: homework 버킷 = 비공개(미성년 음성) · 접근은 signed URL만 · 파일 삭제 정책 없음(폐기=소프트)
```

원안에서 바뀐 것(수리 7건의 스키마 반영): `TIMESTAMPTZ` 문법 교정 · verified `CASCADE→RESTRICT` · 최종 텍스트 1칸→**2칸**(student/corrected) · 버전 3필드+`was_edited` · `task_type`·`privacy_class`·`stt_segments`·`is_audit_sample` · `error_type` 단수→배열 · telemetry=JSONB(5신 채택·0 디폴트 금지) · `l1_source_phrase` 2필드(5신).

**5신 「단일 homework_pipeline 테이블 + ai_draft/verified_data JSONB 병합」은 기각** — ①verified 쌍(두 텍스트)의 NOT NULL 강제가 불가해 금고에 빈 값이 조용히 들어간다 ②모델 업그레이드 후 재처리하면 1행짜리 ai_draft가 덮여 1차안 이력이 사라진다(한수더 파기) ③파인튜닝 추출기가 JSONB 키 오타에 무방비. 「처음부터 단단하게」와 정면 충돌 — 3테이블 유지.

## §2 n8n 워크플로 수정 명세 (유호 JSON 기준 노드별 diff)

| # | 원안 | 수정 | 왜 |
|---|---|---|---|
| 1 | Webhook 트리거 | **Schedule(1~2분) + Select**(`status='pending' and retry_count<3` order by created_at limit 10) → 즉시 `status='processing'` 선점 update | 웹훅 유실 시 pending 고아·재시도 없음. 폴링은 유실·중복·재시도가 공짜 |
| 2 | HTTP GET file_url | Supabase Storage **signed URL** 발급 후 다운로드 | 버킷 비공개(미성년 음성) — 공개 URL 금지 |
| 3 | Whisper 노드 | `response_format: verbose_json`, language ko | segments·avg_logprob 확보 — 무음 하이라이트·검수 레인의 원천 |
| 4 | Gemini 1.5 Flash | **현행 flash(3.6)** · Structured Output 모드(스키마 고정) · `error_type`→배열 | 1.5는 구모델 · `JSON.parse($json.text)`는 마크다운 펜스에 깨짐 |
| 5 | Insert 로그 | + `stt_model_version`·`llm_model_version`·`prompt_version`·`stt_segments`·`stt_avg_logprob` 매핑 | 한수더 — 버전 없는 라벨은 나중에 못 쓴다 |
| 6 | Update status | `ai_processed` + Code 노드 1개로 세그먼트 기반 wpm(음절/초 환산)·정적 초 계산해 캐시 | WPM은 한국어 표준 정의가 없음 → **음절/초**로 통일, 배속 반영은 다음 세션 시작 시 1회 |
| 7 | (없음) | Error Workflow: 실패 시 `status='pending'`, `retry_count+1` | 3회 초과분만 사람 눈에 |

프롬프트 초안(4번 노드, prompt_version `correct-v1`): 입력=stt_raw_text+세그먼트, 출력 스키마 `{corrected_text, error_types[], l1_source_phrase, mongolian_explanation}` — **교정문은 자연스러운 한국어 1문장, 해설은 몽골어, 원문 재서술 금지**. 몽골어 해설 품질은 `tools/몽골어대조.js` 통로로 표본 검증.

## §3 검수 대시보드 스펙 (Expo — talk 앱 내 검수자 역할 화면)

- 진입: role='inspector' 계정만. PC 권장(Expo web) — 키보드 전제.
- 좌: 오디오 플레이어(파형 + `stt_segments` 기반 무음·저신뢰 구간 색 하이라이트) · **텍스트 2칸**: ①「들린 대로」(stt_raw_text 프리필, 편집 가능 — 학생 오류 보존이 목적임을 UI 문구로 명시) ②「교정문」(ai_corrected_text 프리필, 편집 가능)
- 우: 태그 버튼(복수 선택) + 몽골어 해설 초안(클릭 시만 편집)
- 단축키: `Space` 재생/일시정지 · `Enter` 승인+다음 · `1` 직역 · `2` 코드스위칭 구간 지정 · `3` 발음 · `Backspace` 폐기
- **게이트**: 청취 시작 전 `Enter` 무효(Enter 난사 기계 차단) · `listened_ms` 자동 기록 · `was_edited` = 두 텍스트칸 변경 여부로 자동 판정
- 폐기 = 소프트만: `status='discarded'`, 파일 존치(노이즈도 강건성 학습 재료)
- 큐 순서: `is_audit_sample` 우선 혼입 + `stt_avg_logprob` 낮은 순 · **다음 항목 오디오 백그라운드 프리로드**(5신 채택 — 대기 0초)
- 5신 JS 골격은 채택하되 두 가지는 이 발주서가 우선: Enter는 **청취 게이트 통과 후에만** 유효(5신 원안엔 게이트 없음 — 무게이트 승인 기각) · 폐기는 소프트만
- (제안·유호 답 대기) 검수 0단계=학생 본인 확인: 제출 직후 저신뢰 구간만 「이렇게 말한 것 맞아요?」 5초 확인

## §4 수용 기준 (전부 통과해야 완료)

1. 새 Supabase 프로젝트에 §1 DDL 원샷 적용 — 오류 0.
2. 학생 A 계정으로 학생 B 행 select → 0행(RLS 실측).
3. 실제 몽골 억양 20~30초 음성 1건: 제출→`ai_processed` 자동(수동 개입 0) · `stt_segments`·버전 3필드 채워짐.
4. 대시보드 왕복 1건: 수정 승인 → verified 행에 두 텍스트+`was_edited=true`+`listened_ms>0`.
5. 청취 0초 상태에서 Enter → 승인 안 됨(게이트 실측).
6. 폐기 1건: `status='discarded'` 인데 Storage 파일 존치.
7. n8n 죽인 상태로 1건 제출 → 재기동 후 자동 처리(폴링 자가치유 실측).
8. 실기기 2대(iOS/Android) 녹음 파일이 WAV/PCM 16kHz mono인지 파일 헤더로 실측(배선① — 압축·AGC가 신호를 지우면 소급 불가).
9. pg_cron 다음날 접속 0 행 생성 확인.

### 기각·보류 (판정 유지 · 재론 불요)

- 🚫 **노이즈 다양화·하드모드 소음** — 소음은 사후 합성 가능. 수집 항목이 아니라 경험 백로그(2신 ⑨ 판정 유지).
- 🚫 **조음 위치 3D 시각화(혀·입술)** — 단안 카메라엔 기준자가 없어 절대값이 안 나온다. 원순 모음 3쌍·서버측으로 축소된 것이 정본(§7).
- ⏸ **음소 단위 Forced Alignment** — 지금 안 한다. **원본 WAV를 무손실로 보존하므로 언제든 소급 가능**하고, 지금 붙이면 파이프라인만 무거워진다.
- ⏸ 세션 리포트·TTS 3단계 속도·상대 랭킹·신조어 DB·비자 RAG 지식베이스·B2B 산업 데이터셋 — **전부 수집이 아니라 앱 기능/콘텐츠**다. 필요한 원천(어휘 등급·WPM·정적·오류 유형)은 이미 위 스키마에서 파생된다.
- ⏳ **상급생이 몽골어로 해설**(H-1) — 2신 도전안 「선배 학생 유료 마이크로워크」와 같은 항목, 유호 답 대기 중.
- ⏳ **문화적 거부감·정의적 여과막**(H-4) — 검수자가 판정할 수 없는 값(학생 자기보고여야 함). 앱 2단계에서 1문항으로.
